  function createSafeStorage(options = {}) {
    const debug = options.debug;
    const backend = options.storage === void 0 ? defaultStorage(debug) : options.storage;
    let lastError = null;
    function call(method, fallback, args) {
      if (!backend || typeof backend[method] !== "function") {
        lastError = new Error("Storage backend indisponivel.");
        report(debug, "storage-unavailable", { method }, lastError);
        return fallback;
      }
      try {
        const value = backend[method].apply(backend, args);
        lastError = null;
        return value;
      } catch (error) {
        lastError = error;
        report(debug, "storage-" + method + "-failed", { key: args[0] }, error);
        return fallback;
      }
    }
    return {
      get(key) {
        return call("getItem", null, [key]);
      },
      set(key, value) {
        return call("setItem", false, [key, value]) !== false;
      },
      remove(key) {
        return call("removeItem", false, [key]) !== false;
      },
      status() {
        return { available: Boolean(backend), lastError };
      }
    };

      documentRef.addEventListener("DOMContentLoaded", waitForLog, { once: true });
    } else {
      waitForLog();
    }
    return {
      get overlay() {
        return overlay;
      },
      destroy() {
        destroyed = true;
        if (waitingForDomReady) {
          documentRef.removeEventListener("DOMContentLoaded", waitForLog);
          waitingForDomReady = false;
        }
        if (rootObserver) rootObserver.disconnect();
        if (overlay) overlay.destroy();
        if (windowRef.__personChatOverlay === overlay) {
          try {
            delete windowRef.__personChatOverlay;
          } catch {
            windowRef.__personChatOverlay = void 0;
          }
        }
        rootObserver = null;
        overlay = null;
      }
    };
  }
  if (typeof document !== "undefined" && typeof window !== "undefined") {
    const previousBootstrap = window.__personChatBootstrap;
    const previousOverlay = window.__personChatOverlay;
    if (previousBootstrap && typeof previousBootstrap.destroy === "function") previousBootstrap.destroy();
    else if (previousOverlay && typeof previousOverlay.destroy === "function") previousOverlay.destroy();
    window.__personChatBootstrap = bootstrapPersonChat();
  }
