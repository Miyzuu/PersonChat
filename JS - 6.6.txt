// ===== STORAGE =====
function storageGet(key) {
    try { return localStorage.getItem(key); } catch(e) { return null; }
}

function storageSet(key, value) {
    try { localStorage.setItem(key, value); } catch(e) {}
}

function parseJSON(value) {
    if (!value) return null;
    try {
        const parsed = JSON.parse(value);
        return parsed && typeof parsed === "object" ? parsed : null;
    } catch(e) {
        return null;
    }
}

function loadDB(primaryKey, legacyKeys, migrateEntry) {
    const primary = parseJSON(storageGet(primaryKey));
    if (primary) return { db: primary, hadStorage: true };

    for (const legacyKey of legacyKeys) {
        const legacy = parseJSON(storageGet(legacyKey));
        if (!legacy) continue;

        const migrated = {};
        for (const key in legacy) {
            const next = migrateEntry(key, legacy[key]);
            if (next && next.key) migrated[next.key] = next.value;
        }
        storageSet(primaryKey, JSON.stringify(migrated));
        return { db: migrated, hadStorage: true };
    }

    return { db: {}, hadStorage: false };
}

function normalizeToken(value) {
    const token = String(value || "").trim();
    if (!token || token.includes("{") || token.includes("}")) return "";
    return token;
}

function escapeXml(value) {
    return String(value || "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&apos;");
}

function isUsableColor(value) {
    const color = normalizeToken(value);
    return color.startsWith("#") || color.startsWith("rgb") || color.startsWith("hsl");
}

const avatarLoad = loadDB(CONFIG.AVATAR_CACHE_KEY, CONFIG.AVATAR_LEGACY_KEYS, (username, data) => {
    const cleanName = normalizeToken(username).toLowerCase();
    if (!cleanName || !data || !data.url) return null;
    return {
        key: `twitch:${cleanName}`,
        value: { url: data.url, ts: Number(data.ts) || Date.now() }
    };
});

let avatarDB = avatarLoad.db;
let avatarSaveTimer = null;
const pendingAvatarFetches = new Map();

function flushAvatarDB() {
    clearTimeout(avatarSaveTimer);
    avatarSaveTimer = null;
    storageSet(CONFIG.AVATAR_CACHE_KEY, JSON.stringify(avatarDB));
}

function scheduleAvatarSave() {
    clearTimeout(avatarSaveTimer);
    avatarSaveTimer = setTimeout(flushAvatarDB, CONFIG.AVATAR_SAVE_DEBOUNCE);
}

function getAvatarFromDB(userKey) {
    const data = avatarDB[userKey];
    if (data && data.url && Date.now() - data.ts < CONFIG.AVATAR_EXPIRY) return data.url;
    return null;
}

function saveAvatarToDB(userKey, url) {
    if (!userKey || !url) return;

    const current = avatarDB[userKey];
    const now = Date.now();
    if (current && current.url === url && now - current.ts < 6 * 60 * 60 * 1000) return;

    avatarDB[userKey] = { url: url, ts: now };

    const keys = Object.keys(avatarDB);
    if (keys.length > CONFIG.MAX_AVATAR_DB_SIZE) {
        keys.sort((a, b) => (avatarDB[a].ts || 0) - (avatarDB[b].ts || 0));
        for (let i = 0; i < Math.min(100, keys.length - CONFIG.MAX_AVATAR_DB_SIZE); i++) {
            delete avatarDB[keys[i]];
        }
    }

    scheduleAvatarSave();
}

function fetchTwitchAvatar(username) {
    const login = normalizeToken(username).toLowerCase();
    if (!login) return Promise.resolve(null);

    const key = `twitch:${login}`;
    if (pendingAvatarFetches.has(key)) return pendingAvatarFetches.get(key);

    const request = fetch(`https://api.ivr.fi/v2/twitch/user?login=${encodeURIComponent(login)}`)
        .then(res => res.ok ? res.json() : null)
        .then(data => {
            if (Array.isArray(data) && data[0] && data[0].logo) return data[0].logo;
            if (data && data.logo) return data.logo;
            return null;
        })
        .catch(() => null)
        .finally(() => pendingAvatarFetches.delete(key));

    pendingAvatarFetches.set(key, request);
    return request;
}

function fetchKickAvatar(username) {
    const login = normalizeToken(username).toLowerCase();
    if (!login || !CONFIG.FETCH_KICK_AVATAR) return Promise.resolve(null);

    const key = `kick:${login}`;
    if (pendingAvatarFetches.has(key)) return pendingAvatarFetches.get(key);

    const request = fetch(`https://kick.com/api/v2/channels/${encodeURIComponent(login)}`)
        .then(res => res.ok ? res.json() : null)
        .then(data => {
            if (data && data.user && data.user.profile_pic) return data.user.profile_pic;
            if (data && data.profile_pic) return data.profile_pic;
            return null;
        })
        .catch(() => null)
        .finally(() => pendingAvatarFetches.delete(key));

    pendingAvatarFetches.set(key, request);
    return request;
}

const seenLoad = loadDB(CONFIG.SEEN_CACHE_KEY, CONFIG.SEEN_LEGACY_KEYS, (username, value) => {
    const cleanName = normalizeToken(username).toLowerCase();
    if (!cleanName) return null;
    return {
        key: `twitch:${cleanName}`,
        value: value === true ? Date.now() : Number(value) || Date.now()
    };
});

let seenDB = seenLoad.db;
let seenSaveTimer = null;
let firstMessageRadarReady = seenLoad.hadStorage;
let lastVisibleMessage = null;

function cleanupSeenDB() {
    const now = Date.now();
    let changed = false;

    for (const key in seenDB) {
        if (seenDB[key] === true) {
            seenDB[key] = now;
            changed = true;
        } else if (!seenDB[key] || now - Number(seenDB[key]) > CONFIG.SEEN_EXPIRY) {
            delete seenDB[key];
            changed = true;
        }
    }

    const keys = Object.keys(seenDB);
    if (keys.length > CONFIG.MAX_SEEN_DB_SIZE) {
        keys.sort((a, b) => Number(seenDB[a] || 0) - Number(seenDB[b] || 0));
        const removeCount = keys.length - CONFIG.MAX_SEEN_DB_SIZE;
        for (let i = 0; i < removeCount; i++) delete seenDB[keys[i]];
        changed = true;
    }

    if (changed) storageSet(CONFIG.SEEN_CACHE_KEY, JSON.stringify(seenDB));
}

cleanupSeenDB();

if (!firstMessageRadarReady) {
    setTimeout(() => {
        firstMessageRadarReady = true;
    }, CONFIG.FIRST_MESSAGE_WARMUP);
}

function flushSeenDB() {
    clearTimeout(seenSaveTimer);
    seenSaveTimer = null;
    storageSet(CONFIG.SEEN_CACHE_KEY, JSON.stringify(seenDB));
}

function saveSeenDB() {
    clearTimeout(seenSaveTimer);
    seenSaveTimer = setTimeout(flushSeenDB, CONFIG.SEEN_SAVE_DEBOUNCE);
}

function nowTime() {
    return new Date().toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" });
}

function getPlatform(row) {
    const attrs = [
        "data-platform",
        "data-provider",
        "data-service",
        "data-source",
        "data-site",
        "data-channel",
        "data-message-source"
    ];

    for (const attr of attrs) {
        const value = normalizeToken(row.getAttribute(attr)).toLowerCase();
        if (value) return normalizePlatform(value);
    }

    const className = String(row.className || "").toLowerCase();
    if (className.includes("twitch")) return "twitch";
    if (className.includes("youtube") || className.includes("yt-")) return "youtube";
    if (className.includes("kick")) return "kick";

    const avatarUrl = normalizeToken(row.querySelector(".avatar")?.getAttribute("data-avatar") || row.querySelector(".avatar")?.src).toLowerCase();
    if (avatarUrl.includes("twitch") || avatarUrl.includes("jtvnw")) return "twitch";
    if (avatarUrl.includes("ytimg") || avatarUrl.includes("youtube") || avatarUrl.includes("googleusercontent")) return "youtube";
    if (avatarUrl.includes("kick")) return "kick";

    const html = row.innerHTML.toLowerCase();
    if (html.includes("twitch")) return "twitch";
    if (html.includes("youtube") || html.includes("ytimg")) return "youtube";
    if (html.includes("kick")) return "kick";

    return "unknown";
}

function normalizePlatform(value) {
    const platform = String(value || "").toLowerCase();
    if (platform.includes("twitch")) return "twitch";
    if (platform.includes("youtube") || platform === "yt") return "youtube";
    if (platform.includes("kick")) return "kick";
    return platform.replace(/[^a-z0-9_-]/g, "") || "unknown";
}

        </svg>
    `);
}

function isImageUrl(value) {
    try {
        const url = new URL(value);
        return /\.(png|jpe?g|gif|webp)$/i.test(url.pathname);
    } catch(e) {
        return false;
    }
}

function cleanUrlCandidate(value) {
    return String(value || "").replace(/[),.!?\]]+$/g, "");
}

function escapeRegExp(value) {
    return String(value || "").replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}

function embedImages(row) {
    const msgEl = row.querySelector(".user-message");
    if (!msgEl || msgEl.querySelector(".embedded-image-container")) return;

    const text = msgEl.textContent || "";
    const candidates = text.match(/https?:\/\/[^\s<>"']+/gi);
    if (!candidates) return;

    const urls = [...new Set(candidates.map(cleanUrlCandidate).filter(isImageUrl))];
    if (!urls.length) return;

    const container = document.createElement("div");
    container.className = "embedded-image-container";

    urls.forEach(url => {
        const img = document.createElement("img");
        img.src = url;
        img.className = "chat-embedded-image";
        img.loading = "lazy";
        img.referrerPolicy = "no-referrer";
        container.appendChild(img);
    });

    msgEl.appendChild(container);
}

function getNativeFirstFlag(row) {
    const attrs = ["data-first-message", "data-first-msg", "data-is-first"];
    const trueValues = ["1", "true", "yes", "sim", "first", "first-message", "new-chatter"];
    const falseValues = ["0", "false", "no", "nao", "não"];

    for (const attr of attrs) {
        const value = normalizeToken(row.getAttribute(attr)).toLowerCase();
        if (!value) continue;
        if (trueValues.includes(value)) return true;
        if (falseValues.includes(value)) return false;
    }

    return null;
}

function hasPrivilegedBadge(badgesEl) {
    if (!badgesEl) return false;

    const keywords = [
        "subscriber",
        "founder",
        "moderator",
        "mod",
        "vip",
        "broadcaster",
        "streamer",
        "staff",
        "partner"
    ];

    const badgeImgs = badgesEl.querySelectorAll("img, svg");
    for (const img of badgeImgs) {
        const alt = (img.getAttribute("alt") || "").toLowerCase();
        const title = (img.getAttribute("title") || "").toLowerCase();
        const src = (img.getAttribute("src") || "").toLowerCase();
        if (keywords.some(k => alt.includes(k) || title.includes(k) || src.includes(k))) return true;
    }

    const html = badgesEl.innerHTML.toLowerCase();
    return keywords.some(k => html.includes(k));
}

function badgeText(badgesEl) {
    if (!badgesEl) return "";

    const parts = [badgesEl.innerHTML || ""];
    badgesEl.querySelectorAll("img, svg").forEach(el => {
        parts.push(el.getAttribute("alt") || "");
        parts.push(el.getAttribute("title") || "");
        parts.push(el.getAttribute("src") || "");
        parts.push(el.getAttribute("href") || "");
    });

    return parts.join(" ").toLowerCase();
}

function keepIconBadgesOnly(badgesEl) {
    if (!badgesEl) return;

    Array.from(badgesEl.childNodes).forEach(node => {
        if (node.nodeType === Node.TEXT_NODE && node.textContent.trim()) {
            node.remove();
            return;
        }

        if (node.nodeType !== Node.ELEMENT_NODE) return;

        const tag = node.tagName.toLowerCase();
        const hasIcon = tag === "img" ||
            tag === "svg" ||
            Boolean(node.querySelector?.("img, svg"));

        if (!hasIcon) node.remove();
    });
}

function getReplyInfo(row, msgText) {
    const attrs = [
        "data-reply-parent",
        "data-reply-parent-name",
        "data-reply-parent-username",
        "data-reply-parent-display-name",
        "data-reply-parent-user-name",
        "data-reply-to",
        "data-reply-to-name"
    ];

    for (const attr of attrs) {
        const value = normalizeToken(row.getAttribute(attr));
        if (value) return { target: value, fromMentionFallback: false };
    }

    if (!CONFIG.REPLY_FALLBACK_FROM_LEADING_MENTION) {
        return { target: "", fromMentionFallback: false };
    }

    const match = String(msgText || "").match(/^@([^\s:,.!?]+)\s+/);
    if (!match || !match[1]) return { target: "", fromMentionFallback: false };

    return { target: match[1], fromMentionFallback: true };
}

function removeReplyMentionFromMessage(msgEl, replyTarget) {
    if (!msgEl || !replyTarget) return;

    const target = String(replyTarget).replace(/^@/, "");
    const pattern = new RegExp(`^@${escapeRegExp(target)}[\\s:,-]+`, "i");

    for (const node of Array.from(msgEl.childNodes)) {
        if (node.nodeType === Node.TEXT_NODE) {
            const nextText = node.textContent.replace(pattern, "");
            if (nextText !== node.textContent) {
                node.textContent = nextText;
                return;
            }
        }

        if (node.nodeType === Node.ELEMENT_NODE && node.textContent) {
            const nextText = node.textContent.replace(pattern, "");
            if (nextText !== node.textContent) {
                node.textContent = nextText;
                return;
            }
        }
    }

    msgEl.textContent = msgEl.textContent.replace(pattern, "");
}

function applyMessageChain(row, userKey, hasSpecialState) {
    if (!CONFIG.HIDE_REPEATED_AVATAR_AND_NAME || hasSpecialState) {
        lastVisibleMessage = { userKey, ts: Date.now() };
        return;
    }

    const now = Date.now();
    const isRepeated =
        lastVisibleMessage &&
        lastVisibleMessage.userKey === userKey &&
        now - lastVisibleMessage.ts <= CONFIG.REPEATED_MESSAGE_WINDOW;

    if (isRepeated) row.classList.add("msg-compact-chain");

    lastVisibleMessage = { userKey, ts: now };
}

function applyAvatar(row, username, lowerName, userColor, platform, userKey) {
    const avatarEl = row.querySelector(".avatar");
    if (!avatarEl) return;

    const rawAvatar = normalizeToken(avatarEl.getAttribute("data-avatar") || row.getAttribute("data-avatar"));
    const fallbackSvg = generateInitialsAvatar(username, userColor);

    const applyImage = (url) => {
        if (!url) {
            avatarEl.src = fallbackSvg;
            return;
        }

        const img = new Image();
        img.onload = () => {
            if (document.contains(avatarEl)) avatarEl.src = url;
        };
        img.onerror = () => {
            if (document.contains(avatarEl)) avatarEl.src = fallbackSvg;
        };
        img.src = url;
    };

    const cachedUrl = getAvatarFromDB(userKey);
    if (cachedUrl) {
        applyImage(cachedUrl);
        return;
    }

    const hasGoodRawAvatar = rawAvatar &&
        rawAvatar !== TRANSPARENT_PIXEL &&
        !rawAvatar.includes("default") &&
        !rawAvatar.includes("blank");

    if (hasGoodRawAvatar) {
        applyImage(rawAvatar);
        saveAvatarToDB(userKey, rawAvatar);
        return;
    }

    if (platform === "kick") {
        fetchKickAvatar(lowerName).then(url => {
            if (url) {
                applyImage(url);
                saveAvatarToDB(userKey, url);
            } else if (document.contains(avatarEl)) {
                avatarEl.src = fallbackSvg;
            }
        });
        return;
    }

    const canFetchTwitchAvatar = platform === "twitch" ||
        (platform === "unknown" && CONFIG.FETCH_TWITCH_AVATAR_WHEN_PLATFORM_UNKNOWN);

    if (!canFetchTwitchAvatar) {
        avatarEl.src = fallbackSvg;
        return;
    }

    fetchTwitchAvatar(lowerName).then(url => {
        if (url) {
            applyImage(url);
            saveAvatarToDB(userKey, url);
        } else if (document.contains(avatarEl)) {
            avatarEl.src = fallbackSvg;
        }
    });
}

// ===== PROCESSAMENTO PRINCIPAL =====
function processRow(row, options = {}) {
    if (!row || row.dataset.processed === "1") return;
    row.dataset.processed = "1";

    const username = normalizeToken(row.getAttribute("data-from")) || "User";
    const lowerName = username.toLowerCase();
    const platform = getPlatform(row);
    const userKey = getUserKey(platform, lowerName);
    const userColor = normalizeToken(row.getAttribute("data-color")) || "#888";
    const msgEl = row.querySelector(".user-message");
    const msgText = msgEl ? msgEl.textContent.trim() : "";

    row.classList.add(`platform-${platform}`);

    if (CONFIG.HIDE_COMMANDS && msgText.startsWith("!")) {
        row.remove();
        return;
    }

    const badgesEl = row.querySelector(".badges");
    if (badgesEl && badgesEl.innerHTML.includes("{badges}")) {
        badgesEl.innerHTML = badgesEl.innerHTML.replace(/{badges}/g, "");
    }
    keepIconBadgesOnly(badgesEl);

    const timeEl = row.querySelector(".chat-time");
    if (timeEl) timeEl.textContent = nowTime();

    const nameEl = row.querySelector(".user-name");
    if (nameEl && isUsableColor(userColor)) nameEl.style.color = userColor;

    addPlatformBadge(row, platform);

    const roles = getUserRoles(badgesEl, row);
    applyRoleClasses(row, roles);
    addRolePills(row, roles);

    const events = detectMessageEvents(row, msgText);
    addEventBanners(row, events);

    applyAvatar(row, username, lowerName, userColor, platform, userKey);

    let isFirstTimeChatter = false;
    const nativeFirst = getNativeFirstFlag(row);
    const isHistorical = Boolean(options.historical);

    if (!isHistorical) {
        if (nativeFirst === true) {
            isFirstTimeChatter = true;
        } else if (
            nativeFirst !== false &&
            firstMessageRadarReady &&
            !seenDB[userKey] &&
            !hasPrivilegedBadge(badgesEl)
        ) {
            isFirstTimeChatter = true;
        }

        if (row.innerHTML.toLowerCase().includes("new-chatter")) {
            isFirstTimeChatter = true;
        }
    }

    seenDB[userKey] = Date.now();
    saveSeenDB();

    if (isFirstTimeChatter) addFirstMessageBanner(row);

    const replyInfo = getReplyInfo(row, msgText);
    if (replyInfo.target) {
        addReplyPill(row, replyInfo.target);
        removeReplyMentionFromMessage(msgEl, replyInfo.target);
    }

    applyMessageChain(row, userKey, isFirstTimeChatter || Boolean(replyInfo.target) || events.length > 0);

    embedImages(row);

    if (CONFIG.HIDE_AFTER !== 999) {
        row._hideTimer = setTimeout(() => removeWithFade(row), CONFIG.HIDE_AFTER * 1000);
    }

    trimMessages();
}

// ===== OBSERVER =====
function startObserver() {
    const log = document.getElementById("log");
    if (!log) return;

    applyOverlaySettings(log);

    log.querySelectorAll(".message-row").forEach(row => processRow(row, { historical: true }));

    const observer = new MutationObserver(mutations => {
        for (const mutation of mutations) {
            mutation.addedNodes.forEach(node => {
                if (!(node instanceof HTMLElement)) return;

                if (node.classList.contains("message-row")) {
                    processRow(node, { historical: false });
                }

                node.querySelectorAll?.(".message-row").forEach(row => {
                    processRow(row, { historical: false });
                });
            });
        }
    });

    observer.observe(log, { childList: true, subtree: true });
}

window.addEventListener("beforeunload", () => {
    flushAvatarDB();
    flushSeenDB();
});

document.addEventListener("visibilitychange", () => {
    if (document.visibilityState === "hidden") {
        flushAvatarDB();
        flushSeenDB();
    }
});

startObserver();
